cmake_minimum_required(VERSION 3.18)
project(wih_submission_study LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(embree 4 CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
add_executable(wih_study study.cpp)
target_link_libraries(wih_study PRIVATE embree OpenMP::OpenMP_CXX)
target_compile_definitions(wih_study PRIVATE WIH_HAS_OPENMP=1)
target_compile_options(wih_study PRIVATE -O3 -DNDEBUG -Wall -Wextra)
